# Greener Grass Campaign: much stronger migration pull, so only decree states pass MIGRATION_PULL_HIGH_THRESHOLD.

REPLACE:decree_greener_grass_campaign = {
	texture = "gfx/interface/icons/decree/decree_greener_grass_campaign.dds"
	country_trigger = {
		custom_tooltip = {
			text = tsarpoints_decree_not_upgraded_tt
			NOT = { has_variable = tsarpoints_upgraded_greener_grass_campaign }
		}
	}

	state_trigger = {
		is_isolated_from_market = no
	}
	modifier = {
		state_migration_pull_add = 200
		state_migration_pull_mult = 0.25
	}
	unlocking_technologies = {
		romanticism
	}
	cost = 100

	ai_weight = {
		value = 0
		if = {
			limit = {
				any_scope_building = {
					is_subsistence_building = yes
					occupancy < 0.75
				}
			}
			add = 10
		}
		if = {
			limit = {
				any_scope_building = {
					is_subsistence_building = yes
					occupancy < 0.5
				}
			}
			add = 15
		}
		if = {
			limit = {
				any_scope_building = {
					is_subsistence_building = yes
					occupancy < 0.25
				}
			}
			add = 25
		}
		if = {
			limit = {
				scope:country = {
					has_journal_entry = je_the_algerian_departements
				}
				state_region = {
					OR = {
						this = s:STATE_ORAN
						this = s:STATE_ALGIERS
						this = s:STATE_CONSTANTINE
					}
				}
				culture_percent_state = {
					target = cu:french
					value <= 0.25 # Bit of a buffer since we need 20%
				}
			}
			add = 500
		}
		if = {
			limit = {
				is_incorporated = yes
				any_scope_building = {
					is_subsistence_building = yes
					occupancy < 0.9
				}
				scope:country = {
					OR = {
						has_journal_entry = je_south_america_migration
						has_journal_entry = je_central_america_migration
						has_journal_entry = je_american_west_migration
					}
				}
			}
			add = 500
		}
		if = {
			limit = {
				scope:country = {
					has_journal_entry = je_taming_the_north
				}
				state_region = {
					this = s:STATE_HOKKAIDO
				}
			}
			add = 500
		}
		if = {
			limit = {
				scope:country = {
					has_journal_entry = je_colonize_korea
				}
				is_in_geographic_region = geographic_region_korea
			}
			add = 500
		}
		if = {
			limit = {
				scope:country = {
					has_journal_entry = je_russification
				}
				culture_percent_state = {
					target = cu:russian
					value < je_russification_decree_russian_culture_target # set in event_values
				}
				OR = {
					is_in_geographic_region = geographic_region_finland_old
					is_in_geographic_region = geographic_region_baltic_states_old
					is_in_geographic_region = geographic_region_poland_old
				}
			}
			add = 250
		}
		if = {
			limit = {
				scope:country = {
					has_journal_entry = je_tp_tsar_heir_of_alexander
					NOT = { has_variable = je_tp_tsar_heir_of_alexander_button_1_complete }
				}
				state_region = {
					this = s:STATE_OUTER_MANCHURIA
				}
			}
			add = 500
		}
		if = {
			limit = {
				scope:country = {
					has_journal_entry = je_tp_pool_develop_frontier
				}
				OR = {
					is_in_geographic_region = geographic_region_siberia
					is_in_geographic_region = geographic_region_central_asia
					is_in_geographic_region = geographic_region_caucasus_old
				}
			}
			multiply = 1.5
		}
		if = {
			limit = {
				is_incorporated = no
				scope:country = { NOT = { has_law_or_variant = law_type:law_colonial_resettlement } } # Already gives a hefty migration bonus
			}
			multiply = 2.0
		}
		if = {
			limit = {
				state_unemployment_rate >= 0.1
			}
			multiply = 0
		}
	}
}
