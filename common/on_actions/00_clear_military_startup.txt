# common/on_actions/00_clear_military_startup.txt
on_game_started = {
	on_actions = {
		earth_mod_clear_military_on_action
		earth_law_making
	}
}

on_game_started_after_lobby = {
	on_actions = {
		earth_warning_event
		earth_law_making
	}
}

on_power_bloc_formed = {
	on_actions = {
		earth_power_bloc
	}
}

earth_power_bloc = {
	effect = {
		c:UNE ?= {
			if = {
				limit = {
					has_power_bloc = yes
				}
				power_bloc = {
					add_modifier = {
						name = empire_mandate_growth
						years = -1
					}
				}
			}
		}
	}
}

earth_mod_clear_military_on_action = {
	effect = {
		# 1. BASES & ESTABLISHMENTS
		every_state = {
			limit = {
				has_building = building_barrack
			}
			remove_building = building_barrack
		}
		every_state = {
			limit = {
				has_building = building_naval_administration
			}
			remove_building = building_naval_administration
		}
		every_state = {
			limit = {
				has_building = building_naval_fortification
			}
			remove_building = building_naval_fortification
		}
		# 2. MILITARY INDUSTRIES
		every_state = {
			limit = {
				has_building = building_arms_industry
			}
			remove_building = building_arms_industry
		}
		every_state = {
			limit = {
				has_building = building_artillery_foundries
			}
			remove_building = building_artillery_foundries
		}
		every_state = {
			limit = {
				has_building = building_munitions_plant
			}
			remove_building = building_munitions_plant
		}
	}
}

earth_create_humans = {
	effect = {
		every_state = {
			# Loops through every pop group inside that state
			every_scope_pop = {
				# Instantly converts exactly 0.1% of the existing population into humans
				# This perfectly retains their current jobs, factories, and wealth tiers
				change_pop_culture = {
					target = cu:human
					value = 0.001
				}
			}
		}
	}
}

earth_warning_event = {
	effect = {
		every_country = {
			limit = {
				is_player = yes
			}
			trigger_event = {
				id = UNE.1
				popup = yes
			}
		}
	}
}
