UNE_socialist = {
	color = "swr_red"
	possible = {
		c:UNE ?= this
		AND = {
			OR = {
				has_law_or_variant = law_type:law_council_republic
				has_game_rule = SWR_skin
			}
			NOT = {
				OR = {
					has_game_rule = UCF_skin
					has_game_rule = IOM_skin
					has_game_rule = COM_skin
					has_game_rule = EarthKingdom_skin
					has_game_rule = UNE_skin
				}
			}
		}
	}
}
